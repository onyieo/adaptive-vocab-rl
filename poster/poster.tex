% =====================================================================
\begin{column}{\colwidth}

  \begin{block}{The Problem}
    A conversational language tutor must decide at every turn:
    \emph{which vocabulary should the tutor's response target?} Each turn
    presents words the learner has never seen, words being acquired, and
    words due for review. The choice trades off:
    \begin{itemize}
      \item \textbf{Long-term retention} — words must come back before they decay
      \item \textbf{Curriculum pacing} — too many new words at once overwhelms
      \item \textbf{Conversational naturalness} — the tutor's turn must still
            sound like dialogue, not a flashcard drill
    \end{itemize}
    This is a sequential decision problem with delayed, multi-objective
    rewards: the effect of introducing a word now on long-term retention
    isn't observed until future sessions.
  \end{block}

  \begin{block}{Why prior work falls short}
    \textbf{Spaced repetition (FSRS, SM-2)} \cite{fsrs} schedules flashcards
    in isolation — no mechanism for embedding review in natural conversation
    and no joint reasoning about introducing new vocabulary.

    \textbf{RLHF-style methods} \cite{rlhf} train the LLM directly but can't
    reason about long-horizon retention; the LLM is the policy, not a
    scheduler.

    \textbf{Educational RL} \cite{rafferty} works on isolated problems
    (which exercise to assign), never on item selection mediated through a
    generative LLM producing free-form dialogue.

    \textbf{Our contribution}: a hierarchical control framing where RL
    operates over latent linguistic constraints and a frozen LLM executes
    them as natural conversation.
  \end{block}

  \begin{block}{Approach: Hierarchical Control}
    \begin{center}
      \begin{tikzpicture}[
          node distance=1.0cm and 1.3cm,

        \node[smallbox, right=of action] (llm) {Frozen LLM\\(Claude Sonnet)};
        \node[smallbox, below=of llm] (utter) {Japanese\\conversational turn};
        \node[smallbox, below=of action] (sim) {Student\\simulator\\(FSRS)};
        \node[smallbox, below=of policy] (state) {State $s$:\\per-word stats};
        \draw[arrow] (policy) -- (action);
        \draw[arrow] (action) -- (llm);
        \draw[arrow] (llm) -- (utter);
        \draw[arrow] (utter) -- (sim);
        \draw[arrow] (sim) -- (state);
        \draw[arrow] (state) -- (policy);
      \end{tikzpicture}
    \end{center}
    \begin{itemize}
      \item \textbf{State} $s_t$: per-word stability, retrievability,
            difficulty, usage counts; session position
      \item \textbf{Action} $a_t$: structured directive
            $(n_{\text{new}}, n_{\text{active}}, n_{\text{review}})$
            summing to 5 words per turn (21 discrete actions)
      \item \textbf{Reward} $r_t$ (v3): combines coverage growth,
            naturalness penalty for overloading new vocabulary, and
            session-end retention bonus
      \item \textbf{Env}: knowledge-tracing student simulator with FSRS-style
            forgetting; LLM tutor (Claude Sonnet) realizes directives as
            Japanese conversational turns
    \end{itemize}
  \end{block}

\begin{column}{\colwidth}

  \begin{block}{Methods Compared}
    All policies operate on the same 21-action directive space.
    \begin{itemize}
      \item \textbf{Random} — uniform over 21 directives
      \item \textbf{FSRS, FSRS+New, RuleBased} — proposal's scheduling
            heuristic baselines
      \item \textbf{SmartFSRS} — hand-coded scheduler with cognitive-load
            awareness (only introduce when active-due count is low)
      \item \textbf{LLM-prompted} — Claude Haiku reads current state and
            picks a directive (the "no-RL" upper bound for prompt-engineered
            systems)
      \item \textbf{IQL} \cite{iql} — offline RL, our primary method
      \item \textbf{CQL} \cite{cql} — offline RL with conservative penalty
      \item \textbf{BC of PPO} — supervised distillation from the online
            policy (ablation: does offline RL add anything beyond imitation?)
      \item \textbf{PPO} \cite{ppo} — online RL trained from scratch
            (deployment-impractical upper bound)
    \end{itemize}
  \end{block}

  \begin{alertblock}{Key Finding: Reward Design Matters More Than Algorithm}
    Three reward generations, three different stories:
    \begin{itemize}
      \item \textbf{v1 (coverage only)}: $r = \Delta R_{\text{total}}/N$ —
            \emph{trivially won by Random} (uniform spray maximizes coverage)
      \item \textbf{v2 (+ rebalanced buffer)}: drop FSRS/RuleBased from
            behavior policies, add PPO distillation, scale rewards $100\times$
            $\Rightarrow$ \emph{offline RL matches online RL} (CQL 112/57
            vs PPO 117/63)
      \item \textbf{v3 (+ retention-aware reward)}: add naturalness penalty
            and session-end retention bonus $\Rightarrow$ \emph{offline RL
            exceeds online RL on retention} (CQL 99 vs PPO 64)
    \end{itemize}
    \heading{v3 reward}
    $$r_t = \underbrace{\Delta R_{\text{total}}/N}_{\text{coverage}}
            \;-\; \underbrace{\alpha \cdot \max(0, n_{\text{new}} - 1)^2}_{\text{naturalness penalty}}
            \;+\; \underbrace{\lambda \cdot \tfrac{\#\{R{>}0.5\}}{N}}_{\text{session-end retention bonus}}$$
  \end{alertblock}

  \begin{block}{Takeaways and Limitations}
    \heading{What we learned}
    \begin{itemize}
      \item \textbf{Reward design dominates algorithm choice} — v1 vs v3
            reward changed CQL retention from 19 to 99
      \item \textbf{Offline RL is competitive with online RL} when the
            buffer is curated and the reward is aligned with the actual goal
      \item \textbf{Hierarchical RL/LLM control} is a viable framing:
            structured directives are tractable as actions; the LLM handles
            naturalness
    \end{itemize}
    \heading{Honest limitations}
    \begin{itemize}
      \item Simulator dynamics are hand-tuned; no human pilot yet
      \item Cross-judge bias check (OpenAI judge) addresses but doesn't
            eliminate LLM-as-judge concerns
      \item Action space is coarse (21 directives) — finer-grained
            word selection is future work
    \end{itemize}
  \end{block}
